## 🏆 1. 실시간 매매 성적 요약 (Overview)

```dataviewjs
const trades = dv.pages('"03_Areas/트레이딩/20_2026_매매일지"')
  .where(p => p.type === "trade");

const total = trades.length;
const wins = trades.where(p => p.status === "Win").length;
const losses = trades.where(p => p.status === "Loss").length;
const be = trades.where(p => p.status === "BE").length;
const ongoing = trades.where(p => p.status === "진행중").length;
const winRate = (wins + losses) > 0 ? ((wins / (wins + losses)) * 100).toFixed(1) : 0;

const longs = trades.where(p => p.position === "Long");
const shorts = trades.where(p => p.position === "Short");
const longWins = longs.where(p => p.status === "Win").length;
const shortWins = shorts.where(p => p.status === "Win").length;
const longRate = longs.length > 0 ? ((longWins / longs.length) * 100).toFixed(0) : 0;
const shortRate = shorts.length > 0 ? ((shortWins / shorts.length) * 100).toFixed(0) : 0;

const ruleFollowedCount = trades.where(p => p.rule_followed === true).length;
const ruleRate = total > 0 ? ((ruleFollowedCount / total) * 100).toFixed(1) : 0;

dv.paragraph(`
| 🎯 총 거래수 | 🟢 승리 (Win) | 🔴 패배 (Loss) | ⚪ 본절 (BE) | ⏳ 진행중 | 🏆 승률 (Win Rate) | 🛡️ 원칙 준수율 |
| :---: | :---: | :---: | :---: | :---: | :---: | :---: |
| **${total} 회** | **${wins} 회** | **${losses} 회** | **${be} 회** | **${ongoing} 회** | **${winRate}%** | **${ruleRate}%** |
`);

dv.paragraph(`
> [!NOTE] ⚖️ 포지션별 승률 분석
> * **🟢 Long 매수 포지션**: 총 ${longs.length}회 중 승리 ${longWins}회 (승률 **${longRate}%**)
> * **🔴 Short 매도 포지션**: 총 ${shorts.length}회 중 승리 ${shortWins}회 (승률 **${shortRate}%**)
`);
```

## 🌊 2. 파동 예측 패턴별 성적표 (Wave Pattern Performance)
> *내가 어떤 파동 패턴(삼각수렴, 지그재그, 플랫 등)에서 높은 승률을 기록하는지 분석합니다.*

```dataview
TABLE WITHOUT ID
  pred_wave_pattern AS "예측 파동 패턴",
  length(rows) AS "매매 횟수",
  length(filter(rows, (r) => r.status = "Win")) AS "승 (Win)",
  length(filter(rows, (r) => r.status = "Loss")) AS "패 (Loss)",
  length(filter(rows, (r) => r.status = "BE")) AS "본절 (BE)",
  choice(
    length(rows) > 0,
    round((length(filter(rows, (r) => r.status = "Win")) / length(rows)) * 100, 1) + "%",
    "0%"
  ) AS "승률 (%)"
FROM "03_Areas/트레이딩/20_2026_매매일지"
WHERE pred_wave_pattern != null AND pred_wave_pattern != ""
GROUP BY pred_wave_pattern
SORT length(rows) DESC
```

## 🧠 3. 심리 상태 & 원칙 준수 분석 (Psychology & Discipline)

```dataview
TABLE WITHOUT ID
  psychology AS "진입 시 심리",
  length(rows) AS "매매 횟수",
  length(filter(rows, (r) => r.status = "Win")) AS "승리",
  length(filter(rows, (r) => r.status = "Loss")) AS "패배",
  length(filter(rows, (r) => r.rule_followed = true)) AS "원칙 준수 횟수"
FROM "03_Areas/트레이딩/20_2026_매매일지"
WHERE psychology != null AND psychology != ""
GROUP BY psychology
```

## 🚨 4. 손실 & 오답 매매 집중 복기 (Loss & Error Review)
> *오답노트에 박제하여 동일한 실수를 방지해야 할 일지 목록입니다.*

```dataview
TABLE WITHOUT ID
  file.link AS "일지 링크",
  date AS "일자",
  position AS "포지션",
  pred_wave_pattern AS "내 예측",
  result_wave_pattern AS "실제 결과",
  final_diag AS "진단",
  error_category AS "오류 분류",
  rule_violated AS "위반 원칙"
FROM "03_Areas/트레이딩/20_2026_매매일지"
WHERE status = "Loss" OR final_diag = "판단미스" OR final_diag = "뇌동매매" OR final_diag = "손절미스"
SORT date DESC
```
